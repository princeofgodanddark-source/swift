local aimbotActive = false
local aimbotConnections = {}
local aimbotGui = nil

local function StopAimbot()
    for _, conn in pairs(aimbotConnections) do
        if conn and conn.Disconnect then
            pcall(function() conn:Disconnect() end)
        end
    end
    aimbotConnections = {}

    if aimbotGui then
        pcall(function() aimbotGui:Destroy() end)
        aimbotGui = nil
    end

    getgenv().CamlockState = false
    getgenv().enemy = nil
end

local function StartAimbot()
    if aimbotActive then
        StopAimbot()
    end

    getgenv().CamlockState = false
    getgenv().enemy = nil
    getgenv().Key = "z"

    local Players = game:GetService("Players")
    local RunService = game:GetService("RunService")
    local LocalPlayer = Players.LocalPlayer
    local Mouse = LocalPlayer:GetMouse()

    local function FindNearestEnemy()
        local ClosestDistance, ClosestPlayer = math.huge, nil
        local CenterPosition = Vector2.new(
            game:GetService("GuiService"):GetScreenResolution().X / 2,
            game:GetService("GuiService"):GetScreenResolution().Y / 2
        )
        for _, Player in ipairs(Players:GetPlayers()) do
            if Player ~= LocalPlayer then
                local Character = Player.Character
                if Character and Character:FindFirstChild("HumanoidRootPart") and Character.Humanoid.Health > 0 then
                    local Position, IsVisibleOnViewport = workspace.CurrentCamera:WorldToViewportPoint(Character.HumanoidRootPart.Position)
                    if IsVisibleOnViewport then
                        local Distance = (CenterPosition - Vector2.new(Position.X, Position.Y)).Magnitude
                        if Distance < ClosestDistance then
                            ClosestPlayer = Character.HumanoidRootPart
                            ClosestDistance = Distance
                        end
                    end
                end
            end
        end
        return ClosestPlayer
    end

    local heartbeatConn
    heartbeatConn = RunService.Heartbeat:Connect(function()
        if getgenv().CamlockState and getgenv().enemy then
            local camera = workspace.CurrentCamera
            local pred = 0.16
            camera.CFrame = CFrame.new(camera.CFrame.p, getgenv().enemy.Position + getgenv().enemy.Velocity * pred)
        end
    end)
    table.insert(aimbotConnections, heartbeatConn)

    local keyConn
    keyConn = Mouse.KeyDown:Connect(function(k)
        if k == getgenv().Key then
            getgenv().CamlockState = not getgenv().CamlockState
            if getgenv().CamlockState then
                getgenv().enemy = FindNearestEnemy()
            else
                getgenv().enemy = nil
            end
        end
    end)
    table.insert(aimbotConnections, keyConn)

    local BladLock = Instance.new("ScreenGui")
    BladLock.Name = "BladLock"
    BladLock.Parent = game.CoreGui
    BladLock.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    aimbotGui = BladLock

    Frame.BorderColor3 = Color3.fromRGB(0, 0, 0)
    Frame.BorderSizePixel = 0
    Frame.Size = UDim2.new(0, 120, 0, 40)
    Frame.Active = true
    Frame.Draggable = true

    local function CenterFrame()
        Frame.Position = UDim2.new(0.5, -Frame.AbsoluteSize.X / 2, 0, -Frame.AbsoluteSize.Y / 2)
    end
    CenterFrame()
    Frame:GetPropertyChangedSignal("AbsoluteSize"):Connect(CenterFrame)

    local UICorner = Instance.new("UICorner")
    UICorner.Parent = Frame

    local Logo = Instance.new("ImageLabel")
    Logo.Name = "Logo"
    Logo.Parent = Frame
    Logo.BackgroundColor3 = Color3.fromRGB(255, 255, 255)

local espEnabled = false
local espBillboards = {}
local espConnections = {}

local function DestroyAllESP()
    for _, billList in pairs(espBillboards) do
        for _, bill in pairs(billList) do
            pcall(function() bill:Destroy() end)
        end
    end
    espBillboards = {}

    for _, conn in pairs(espConnections) do
        pcall(function() conn:Disconnect() end)
    end
    espConnections = {}
end

local function SetupESPForPlayer(player)
    if player == game.Players.LocalPlayer then return end

    local function CreateBillboard(char)
        local head = char:FindFirstChild("Head")
        local humanoid = char:FindFirstChild("Humanoid")
        if not head or not humanoid then
            head = char:WaitForChild("Head", 3)
            humanoid = char:WaitForChild("Humanoid", 3)
            if not head or not humanoid then return end
        end

        local existing = espBillboards[player]
        if existing then
            for i, b in pairs(existing) do
                if b and b.Adornee and b.Adornee.Parent == char then
                    pcall(function() b:Destroy() end)
                    table.remove(existing, i)
                    break
                end
            end
        end

        local billboard = Instance.new("BillboardGui")
        billboard.Size = UDim2.new(0, 60, 0, 20)
        billboard.StudsOffset = Vector3.new(0, 2, 0)
        billboard.Adornee = head
        billboard.Parent = head

        local textLabel = Instance.new("TextLabel")
        textLabel.Size = UDim2.new(1, 0, 1, 0)
        textLabel.BackgroundTransparency = 1
        textLabel.TextScaled = true
        textLabel.Font = Enum.Font.SourceSansBold
        textLabel.TextColor3 = Color3.new(1, 1, 1)
        textLabel.TextStrokeTransparency = 0
        textLabel.TextStrokeColor3 = Color3.new(0, 0, 0)
        textLabel.Text = "100"
        textLabel.Parent = billboard

        if not espBillboards[player] then espBillboards[player] = {} end
        table.insert(espBillboards[player], billboard)

        local function updateHealth(health)
            if health < 0 then health = 0 end
            textLabel.Text = tostring(math.floor(health))
            if health > 50 then
                textLabel.TextColor3 = Color3.new(0, 1, 0)
            elseif health > 25 then
                textLabel.TextColor3 = Color3.new(1, 1, 0)
            else
                textLabel.TextColor3 = Color3.new(1, 0, 0)
            end
        end

        local healthConn = humanoid.HealthChanged:Connect(updateHealth)
        updateHealth(humanoid.Health)

        local function cleanup()
            local list = espBillboards[player]
            if list then
                for i, b in pairs(list) do
                    if b == billboard then
                        table.remove(list, i)
                        break
                    end
                end
            end
            pcall(function() billboard:Destroy() end)
            pcall(function() healthConn:Disconnect() end)
            pcall(function() diedConn:Disconnect() end)
            pcall(function() charAncestryConn:Disconnect() end)
        end

        local diedConn = humanoid.Died:Connect(cleanup)
        local charAncestryConn
        charAncestryConn = char.AncestryChanged:Connect(function()
            if not char.Parent then cleanup() end
        end)
    end

    if player.Character then
        CreateBillboard(player.Character)
    end

    local charAddedConn = player.CharacterAdded:Connect(function(char)
        CreateBillboard(char)
    end)
    table.insert(espConnections, charAddedConn)
end

local function StartESP()
    if espEnabled then return end
    local Players = game:GetService("Players")
    local LocalPlayer = Players.LocalPlayer

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            SetupESPForPlayer(player)
        end
    end

    local playerAddedConn = Players.PlayerAdded:Connect(function(player)
        if player ~= LocalPlayer then
            SetupESPForPlayer(player)
        end
    end)
    table.insert(espConnections, playerAddedConn)

    espEnabled = true
end

local function StopESP()
    DestroyAllESP()
    espEnabled = false
end

MainTab:Toggle({

    Locked = false,
    Default = false,
    Callback = function(state)
        if state then
            StartESP()
        else
            StopESP()
        end
    end
})

local tracerEnabled = false
local tracerBeams = {}
local tracerConnections = {}

local function DestroyAllTracers()
    for _, beam in pairs(tracerBeams) do
        if beam and beam.Parent then
            pcall(function() beam:Destroy() end)
        end
    end
    tracerBeams = {}
    for _, conn in pairs(tracerConnections) do
        pcall(function() conn:Disconnect() end)
    end
    tracerConnections = {}
end

local function CreateTracerForPlayer(player)
    if player == game.Players.LocalPlayer then return end

    local function updateTracer()
        local LocalChar = game.Players.LocalPlayer.Character
        local PlayerChar = player.Character
        if not LocalChar or not PlayerChar then return end
        local localRoot = LocalChar:FindFirstChild("HumanoidRootPart")
        local playerRoot = PlayerChar:FindFirstChild("HumanoidRootPart")
        if not localRoot or not playerRoot then return end

        if tracerBeams[player] and tracerBeams[player].Parent then
            pcall(function() tracerBeams[player]:Destroy() end)
            tracerBeams[player] = nil
        end

        local att0 = Instance.new("Attachment")
        att0.Parent = localRoot
        local att1 = Instance.new("Attachment")
        att1.Parent = playerRoot

        local beam = Instance.new("Beam")
        beam.Name = "ESPBeam"
        beam.Attachment0 = att0
        beam.Attachment1 = att1
        beam.Color = ColorSequence.new(Color3.fromRGB(255, 0, 0))
        beam.FaceCamera = true
        beam.Width0 = 0.15
        beam.Width1 = 0.15
        beam.Parent = player.Character or player
        tracerBeams[player] = beam

        beam.AncestryChanged:Connect(function()
            if not beam.Parent then
                pcall(function() att0:Destroy() end)
                pcall(function() att1:Destroy() end)
                tracerBeams[player] = nil
            end
        end)
    end

    if player.Character then
        updateTracer()
    end

    local charConn = player.CharacterAdded:Connect(function()
        task.wait(0.2)
        updateTracer()
    end)
    table.insert(tracerConnections, charConn)

    local localCharConn = game.Players.LocalPlayer.CharacterAdded:Connect(function()
        task.wait(0.2)
        updateTracer()
    end)
    table.insert(tracerConnections, localCharConn)
end

local function StartTracers()
    if tracerEnabled then return end
    local Players = game:GetService("Players")
    local LocalPlayer = Players.LocalPlayer

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            CreateTracerForPlayer(player)
        end
    end

    local playerAddedConn = Players.PlayerAdded:Connect(function(player)
        if player ~= LocalPlayer then
            CreateTracerForPlayer(player)
        end
    end)
    table.insert(tracerConnections, playerAddedConn)

    tracerEnabled = true
end

local function StopTracers()
    DestroyAllTracers()
    tracerEnabled = false
end

MainTab:Toggle({
    Title = "ESP Tracer",
    Desc = "Draw red beams from you to every enemy",
    Locked = false,
    Default = false,
    Callback = function(state)
        if state then
            StartTracers()
        else
            StopTracers()
        end
    end
})

local isInvisible = false
local realChar = nil
local pseudoAnchor = nil
local invisConnections = {}
local renderConn = nil

local function CleanupInvis()
    if isInvisible and realChar and fakeChar then
        local StoredCF = fakeChar.HumanoidRootPart.CFrame
        fakeChar.HumanoidRootPart.CFrame = realChar.HumanoidRootPart.CFrame
        realChar.HumanoidRootPart.CFrame = StoredCF
        fakeChar.Humanoid:UnequipTools()
        game.Players.LocalPlayer.Character = realChar
        workspace.CurrentCamera.CameraSubject = realChar.Humanoid
        pseudoAnchor = fakeChar.HumanoidRootPart
        for i, v in pairs(fakeChar:GetChildren()) do
            if v:IsA("LocalScript") then
                v.Disabled = true
            end
        end
        isInvisible = false
    end

    if fakeChar then
        pcall(function() fakeChar:Destroy() end)
        fakeChar = nil
    end
    if invisPart then
        pcall(function() invisPart:Destroy() end)
        invisPart = nil
    end

    for _, conn in pairs(invisConnections) do
        pcall(function() conn:Disconnect() end)
    end
    invisConnections = {}
    if renderConn then
        pcall(function() renderConn:Disconnect() end)
        renderConn = nil
    end

    local player = game.Players.LocalPlayer
    if realChar and realChar.Parent == workspace then
        player.Character = realChar
        workspace.CurrentCamera.CameraSubject = realChar:FindFirstChild("Humanoid")
    end
    realChar = nil
    pseudoAnchor = nil
    invisActive = false
end

local function SetupInvis()
    if invisActive then
        CleanupInvis()
    end

    local player = game.Players.LocalPlayer
    realChar = player.Character or player.CharacterAdded:Wait()
    if not realChar then return end

    realChar.Archivable = true
    fakeChar = realChar:Clone()
    invisPart = Instance.new("Part", workspace)
    invisPart.Anchored = true
    invisPart.Size = Vector3.new(200, 1, 200)
    invisPart.CFrame = CFrame.new(0, -500, 0)
    invisPart.CanCollide = true
    fakeChar.Parent = workspace
    fakeChar.HumanoidRootPart.CFrame = invisPart.CFrame * CFrame.new(0, 5, 0)

    for i, v in pairs(fakeChar:GetChildren()) do
        if v:IsA("LocalScript") then
            local clone = v:Clone()
            clone.Disabled = true
            clone.Parent = fakeChar
        end
    end

    for i, v in pairs(fakeChar:GetDescendants()) do
        if v:IsA("BasePart") then
            v.Transparency = 0.7
        end
    end

    local function OnCharacterDied()
        if realChar then
            realChar:Destroy()
            realChar = nil
        end
        if fakeChar then
            fakeChar:Destroy()
            fakeChar = nil
        end
        isInvisible = false
    end

    local function OnCharacterAppearanceLoaded(newChar)
        realChar = newChar
        realChar.Archivable = true
        fakeChar = realChar:Clone()
        invisPart:Destroy()
        invisPart = Instance.new("Part", workspace)
        invisPart.Anchored = true
        invisPart.Size = Vector3.new(200, 1, 200)
        invisPart.CFrame = CFrame.new(0, -500, 0)
        invisPart.CanCollide = true
        fakeChar.Parent = workspace
        fakeChar.HumanoidRootPart.CFrame = invisPart.CFrame * CFrame.new(0, 5, 0)

        for i, v in pairs(fakeChar:GetChildren()) do
            if v:IsA("LocalScript") then
                local clone = v:Clone()
                clone.Disabled = true
                clone.Parent = fakeChar
            end
        end
        for i, v in pairs(fakeChar:GetDescendants()) do
            if v:IsA("BasePart") then
                v.Transparency = 0.7
            end
        end

        local hum = realChar:FindFirstChild("Humanoid")
        if hum then
            local diedConn = hum.Died:Connect(OnCharacterDied)
            table.insert(invisConnections, diedConn)
        end
        local appearanceConn = player.CharacterAppearanceLoaded:Connect(OnCharacterAppearanceLoaded)
        table.insert(invisConnections, appearanceConn)

        if invisActive then
            local StoredCF = realChar.HumanoidRootPart.CFrame
            realChar.HumanoidRootPart.CFrame = fakeChar.HumanoidRootPart.CFrame
            fakeChar.HumanoidRootPart.CFrame = StoredCF
            realChar.Humanoid:UnequipTools()
            player.Character = fakeChar
            workspace.CurrentCamera.CameraSubject = fakeChar.Humanoid
            pseudoAnchor = realChar.HumanoidRootPart
            for i, v in pairs(fakeChar:GetChildren()) do
                if v:IsA("LocalScript") then
                    v.Disabled = false
                end
            end
            isInvisible = true
        end
    end

    local hum = realChar:FindFirstChild("Humanoid")
    if hum then
        local diedConn = hum.Died:Connect(OnCharacterDied)
        table.insert(invisConnections, diedConn)
    end
    local appearanceConn = player.CharacterAppearanceLoaded:Connect(OnCharacterAppearanceLoaded)
    table.insert(invisConnections, appearanceConn)

    renderConn = game:GetService("RunService").RenderStepped:Connect(function()
        if pseudoAnchor then
            pseudoAnchor.CFrame = invisPart.CFrame * CFrame.new(0, 5, 0)
        end
        if fakeChar and fakeChar:FindFirstChild("Humanoid") then
            fakeChar.Humanoid:ChangeState(Enum.HumanoidStateType.Physics)
        end
    end)

    if realChar and fakeChar then
        local StoredCF = realChar.HumanoidRootPart.CFrame
        realChar.HumanoidRootPart.CFrame = fakeChar.HumanoidRootPart.CFrame
        fakeChar.HumanoidRootPart.CFrame = StoredCF
        realChar.Humanoid:UnequipTools()
        player.Character = fakeChar
        workspace.CurrentCamera.CameraSubject = fakeChar.Humanoid
        pseudoAnchor = realChar.HumanoidRootPart
        for i, v in pairs(fakeChar:GetChildren()) do
            if v:IsA("LocalScript") then
                v.Disabled = false
            end
        end
        isInvisible = true
    end

    invisActive = true
end

MainTab:Toggle({
    Title = "Invisible Block",
    Desc = "Make yourself invisible (swap with a dummy)",
    Locked = false,
    Default = false,

        if state then

-- 10. ESP Death Counter
-- ============================================================
local espDeathCounterActive = false
local espDeathCounterConnection = nil
local espDeathCounterState = {}

local function StartESPDeathCounter()
    if espDeathCounterActive then return end
    espDeathCounterActive = true

    local Players = game:GetService("Players")
    local RunService = game:GetService("RunService")
    local player = Players.LocalPlayer

    local strongSkills = {
        ["Omni Directional Punch"] = true,
        ["Death Counter"] = true,
        ["Serious Punch"] = true,
        ["Table Flip"] = true
    }
    local weakSkills = {
        ["Consecutive Punches"] = true,
        ["Normal Punch"] = true,
        ["Shove"] = true,
        ["Uppercut"] = true
    }

    local function createBillboard(target, text)
        if not (target and target:FindFirstChild("Head")) then return end
        local bb = target.Head:FindFirstChild("SkillTag") or Instance.new("BillboardGui")
        bb.Name = "SkillTag"
        bb.Size = UDim2.new(0,100,0,40)
        bb.StudsOffset = Vector3.new(0,2.5,0)
        bb.Adornee = target.Head
        bb.AlwaysOnTop = true
        if not bb.Parent then bb.Parent = target.Head end

        local label = bb:FindFirstChild("TextLabel") or Instance.new("TextLabel", bb)
        label.Size = UDim2.new(1,0,1,0)
        label.BackgroundTransparency = 1
        label.Font = Enum.Font.GothamBold
        label.TextScaled = true
        label.TextColor3 = Color3.new(1,1,1)
        label.TextStrokeTransparency = 0.5
        label.Text = text
    end

    local function removeBillboard(target)
        if target and target:FindFirstChild("Head") and target.Head:FindFirstChild("SkillTag") then
            target.Head.SkillTag:Destroy()
        end
    end

    local function getSkillType(backpack)
        for _, tool in ipairs(backpack:GetChildren()) do
            if strongSkills[tool.Name] then return "strong" end
            if weakSkills[tool.Name] then return "weak" end
        end
        return nil
    end

    espDeathCounterConnection = RunService.Heartbeat:Connect(function()
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= player then
                local char = plr.Character
                local backpack = plr:FindFirstChildOfClass("Backpack")
                if char and backpack then
                    local skillType = getSkillType(backpack)
                    local lastState = espDeathCounterState[plr]

                    if not lastState then
                        espDeathCounterState[plr] = skillType
                        if skillType == "strong" then
                            createBillboard(char, "💢")
                        else
                            removeBillboard(char)
                        end
                    else
                        if skillType == "strong" then
                            if lastState ~= "strong" then
                                createBillboard(char, "💢")
                            end
                            espDeathCounterState[plr] = "strong"
                        elseif skillType == "weak" and lastState == "strong" then
                            createBillboard(char, "☠")
                            espDeathCounterState[plr] = "weak"
                            task.delay(math.random(8,9), function()
                                if espDeathCounterState[plr] == "weak" then
                                    removeBillboard(char)
                                end
                            end)
                        end
                    end
                end
            end
        end
    end)
end

local function StopESPDeathCounter()
    if not espDeathCounterActive then return end
    espDeathCounterActive = false

    if espDeathCounterConnection then
        espDeathCounterConnection:Disconnect()
        espDeathCounterConnection = nil
    end

    for _, plr in ipairs(game:GetService("Players"):GetPlayers()) do
        if plr.Character and plr.Character:FindFirstChild("Head") then
            local head = plr.Character.Head
            local tag = head:FindFirstChild("SkillTag")
            if tag then tag:Destroy() end
        end
    end
    espDeathCounterState = {}
end

MainTab:Toggle({
    Title = "ESP Death Counter",
    Desc = "Shows icons above players with ult (💢) or death counter (☠) skills",
    Locked = false,
    Default = false,
    Callback = function(state)
        if state then
            StartESPDeathCounter()
        else
            StopESPDeathCounter()
        end
    end
})

                            and workspace.Cutscenes:FindFirstChild("Death Cutscene")
                            and workspace.Cutscenes["Death Cutscene"]:FindFirstChild("Camm")

            if deathCam and game:GetService("Workspace").CurrentCamera.CameraSubject == deathCam then
                antiDeathChecking = true
                local player = game.Players.LocalPlayer

                    local originalCFrame = hrp.CFrame

                    if game:GetService("Workspace").CurrentCamera.CameraSubject == deathCam then

                    end
                end
                antiDeathChecking = false
            end
        end
    end)

-- ============================================================
local speedBoostActive = false
local speedBoostConnection = nil

local function StartSpeedBoost()
    if speedBoostActive then return end
    speedBoostActive = true

    local RunService = game:GetService("RunService")
    local player = game.Players.LocalPlayer

    speedBoostConnection = RunService.Heartbeat:Connect(function()
        if speedBoostActive then
            local char = player.Character
            local hum = char and char:FindFirstChild("Humanoid")
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            if hum and hrp and hum.MoveDirection.Magnitude > 0 then
                hrp.CFrame = hrp.CFrame + (hum.MoveDirection * 0.08)
            end
        end
    end)
end

local function StopSpeedBoost()
    speedBoostActive = false
    if speedBoostConnection then
        speedBoostConnection:Disconnect()
        speedBoostConnection = nil
    end
end

MainTab:Toggle({
    Title = "Speed Boost",
    Desc = "Slightly increases movement speed (subtle 0.08x bonus)",
    Locked = false,
    Default = false,
    Callback = function(state)
        if state then
            StartSpeedBoost()
        else
            StopSpeedBoost()
        end
    end
})

-- ============================================================
local jumpBoostActive = false

local function ApplyJumpSettings()
    local player = game.Players.LocalPlayer
    local char = player.Character
    local hum = char and char:FindFirstChild("Humanoid")
    if not hum then return end

    if jumpBoostActive then
        hum.UseJumpPower = true
        hum.JumpHeight = 50
    else
        hum.UseJumpPower = false
        hum.JumpHeight = 7.2
    end
end

game.Players.LocalPlayer.CharacterAdded:Connect(function()
    task.wait(0.1)
    ApplyJumpSettings()
end)

local function StartJumpBoost()
    jumpBoostActive = true
    ApplyJumpSettings()
end

local function StopJumpBoost()
    jumpBoostActive = false
    ApplyJumpSettings()
end

MainTab:Toggle({
    Title = "Jump Boost",
    Desc = "Enables high jump (fixed 50 studs)",
    Locked = false,
    Default = false,
    Callback = function(state)
        if state then
            StartJumpBoost()
        else
            StopJumpBoost()
        end
    end
})

local function disableCollisions(character)
    for _, part in ipairs(character:GetDescendants()) do
        if part:IsA("BasePart") then
            part.CanCollide = false
        end
    end
end

local function getClosestTarget(rootPart, range)
    local closest = nil
    local closestDist = range
    for _, player in ipairs(game.Players:GetPlayers()) do
        if player ~= game.Players.LocalPlayer then
            local char = player.Character
            if char and char:FindFirstChild("HumanoidRootPart") then
                local hrp = char.HumanoidRootPart
                local dist = (hrp.Position - rootPart.Position).Magnitude
                if dist < closestDist then
                    closestDist = dist
                    closest = {Model = char, Root = hrp}
                end
            end
        end
    end
    if closest then
        return closest.Model, closest.Root
    end
    return nil, nil
end

    Locked = false,

                    "rbxassetid://13379003796"
                }
                
                local CONFIG = {
                    TARGET_RANGE = 50,
                    TRACKING_DURATION = 0.45,
                    DASH_DELAY = 0.28,
                    DASH_ACTIVATION_DELAY = 0.22,
                    RESPONSIVENESS = 200,
                    UPPERCUT_BOOST = 20
                }
                
                local LocalPlayer = game.Players.LocalPlayer
                local RunService = game:GetService("RunService")
                local currentCharacter = LocalPlayer.Character
                local currentHumanoid = nil
                local currentRootPart = nil
                local animationConnection = nil
                local trackingConnection = nil
                
                local function setupCharacter(character)
                    currentCharacter = character
                    currentHumanoid = character:WaitForChild("Humanoid")
                    currentRootPart = character:WaitForChild("HumanoidRootPart")
                    
                    if animationConnection then
                        animationConnection:Disconnect()
                    end
                    
                    animationConnection = currentHumanoid.AnimationPlayed:Connect(function(track)
                        local animation = track.Animation
                        if animation and table.find(ANIMATION_IDS, animation.AnimationId) then
                            disableCollisions(currentCharacter)
                            
                            task.wait(CONFIG.DASH_DELAY)
                            
                            local communicate = currentCharacter:FindFirstChild("Communicate")
                            if communicate then
                                communicate:FireServer({["Goal"] = "KeyPress", ["Key"] = Enum.KeyCode.Q})
                            end
                            
                            task.wait(CONFIG.DASH_ACTIVATION_DELAY)
                            
                            local upBoost = Instance.new("BodyVelocity")
                            upBoost.MaxForce = Vector3.new(0, math.huge, 0)
                            upBoost.Velocity = Vector3.new(0, CONFIG.UPPERCUT_BOOST, 0)
                            upBoost.Parent = currentRootPart
                            task.delay(0.2, function()
                                if upBoost then upBoost:Destroy() end
                            end)
                            
                            if currentRootPart then
                                local alignAttachment = Instance.new("Attachment")
                                alignAttachment.Name = "AutoAimAttachment"
                                alignAttachment.Parent = currentRootPart
                                
                                local alignOrientation = Instance.new("AlignOrientation")
                                alignOrientation.Mode = Enum.OrientationAlignmentMode.OneAttachment
                                alignOrientation.Attachment0 = alignAttachment
                                alignOrientation.MaxTorque = math.huge
                                alignOrientation.Responsiveness = CONFIG.RESPONSIVENESS
                                alignOrientation.RigidityEnabled = false
                                alignOrientation.Parent = currentRootPart
                                
                                local bodyVelocity = Instance.new("BodyVelocity")
                                bodyVelocity.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
                                bodyVelocity.Velocity = Vector3.zero
                                bodyVelocity.Parent = currentRootPart
                                
                                local startTime = tick()
                                
                                if trackingConnection then
                                    trackingConnection:Disconnect()
                                end
                                
                                trackingConnection = RunService.Heartbeat:Connect(function()
                                    if CONFIG.TRACKING_DURATION >= tick() - startTime then
                                        local targetModel, targetRoot = getClosestTarget(
                                            currentRootPart,
                                            CONFIG.TARGET_RANGE
                                        )
                                        
                                        if targetModel and targetRoot then
                                            local direction = (targetRoot.Position - currentRootPart.Position)
                                            direction = Vector3.new(direction.X, 0, direction.Z)
                                            
                                            if direction.Magnitude > 0.1 then
                                                local lookCFrame = CFrame.lookAt(
                                                    currentRootPart.Position,
                                                    Vector3.new(targetRoot.Position.X, currentRootPart.Position.Y, targetRoot.Position.Z)
                                                )
                                                
                                                local tiltedCFrame = lookCFrame * CFrame.Angles(math.rad(15), 0, 0)
                                                
                                                currentRootPart.CFrame = tiltedCFrame
                                                alignOrientation.CFrame = tiltedCFrame
                                                
                                                local distance = (targetRoot.Position - currentRootPart.Position).Magnitude
                                                if distance > 3 then
                                                    local velocityDirection = (targetRoot.Position - currentRootPart.Position).Unit
                                                    bodyVelocity.Velocity = velocityDirection * 50
                                                else
                                                    bodyVelocity.Velocity = Vector3.zero
                                                end
                                            end
                                        end
                                    else
                                        if trackingConnection then
                                            trackingConnection:Disconnect()
                                        end
                                        bodyVelocity:Destroy()
                                        alignOrientation:Destroy()
                                        alignAttachment:Destroy()
                                    end
                                end)
                            end
                        end
                    end)
                end
                
                local characterAddedConnection = LocalPlayer.CharacterAdded:Connect(setupCharacter)
                if LocalPlayer.Character then
                    setupCharacter(LocalPlayer.Character)
                end
                
                while toggles.HexedTech do
                    task.wait(1)
                end
                
                if animationConnection then
                    animationConnection:Disconnect()
                end
                if trackingConnection then
                    trackingConnection:Disconnect()
                end
                characterAddedConnection:Disconnect()
            end)
        end
    end
})

local function LoadExternalTech(url, name)
    task.spawn(function()
        pcall(function()
            for _, v in ipairs(game:GetService("Players").LocalPlayer.Character:GetDescendants()) do
                if v:IsA("Animation") or v:IsA("Animator") then
                    pcall(function() v:Stop() end)
                end
            end
            pcall(function()
                for _, v in ipairs(game.CoreGui:GetChildren()) do
                    if v:IsA("ScreenGui") and (v.Name:lower():find("intro") or v.Name:lower():find("welcome") or v.Name:lower():find("loading")) then
                        v:Destroy()
                    end
                end
            end)
        end)
        
        local success, result = pcall(function()
            return loadstring(game:HttpGet(url))()
        end)
        if not success then
            warn(name .. " failed to load:", result)
            game.StarterGui:SetCore("SendNotification", {
                Title = "Error",
                Text = name .. " failed to load!",
                Duration = 3,
            })
        else
            game.StarterGui:SetCore("SendNotification", {
                Title = "Success",
                Text = name .. " loaded successfully!",
                Duration = 2,
            })
        end
    end)
end

    local player = game.Players.LocalPlayer

    if char and char:FindFirstChild("HumanoidRootPart") then
        char.HumanoidRootPart.CFrame = CFrame.new(position)

    end
